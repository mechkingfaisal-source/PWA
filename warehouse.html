function computeStockStatus(stock: number, minStock: number): "normal" | "low" | "critical" | "out" {
  if (stock <= 0) return "out";
  if (stock <= Math.max(1, Math.ceil(minStock * 0.5))) return "critical";
  if (stock <= minStock) return "low";
  return "normal";
}

function computeSuggestedOrderQty(stock: number, minStock: number): number {
  return Math.max(minStock * 2 - stock, minStock, 1);
}

function formatMaterial(m: typeof materialsTable.$inferSelect) {
  return {
    ...m,
    stockStatus: computeStockStatus(m.stock, m.minStock),
    suggestedOrderQty: computeSuggestedOrderQty(m.stock, m.minStock),
  };
}

router.post("/", async (req, res) => {
  try {
    const { type, materialId, qty, user, workOrder, note } = req.body;
    if (!type || !materialId || !qty || !user) {
      res.status(400).json({ error: "type, materialId, qty, and user are required" });
      return;
    }
    
    const [material] = await db.select().from(materialsTable).where(eq(materialsTable.id, materialId));
    if (!material) { res.status(404).json({ error: "Material not found" }); return; }
    
    // Update stock
    let newStock = material.stock;
    const now = new Date().toISOString().slice(0, 10);
    
    if (type === "in") {
      newStock = material.stock + qty;
      await db.update(materialsTable).set({ stock: newStock, lastReceived: now }).where(eq(materialsTable.id, materialId));
    } else if (type === "out") {
      if (material.stock < qty) {
        res.status(400).json({ error: "Insufficient stock" });
        return;
      }
      newStock = material.stock - qty;
      await db.update(materialsTable).set({ stock: newStock, lastIssued: now }).where(eq(materialsTable.id, materialId));
    } else if (type === "count" || type === "adjust") {
      newStock = qty;
      await db.update(materialsTable).set({ stock: newStock, lastCount: now }).where(eq(materialsTable.id, materialId));
    }
    
    const [created] = await db.insert(transactionsTable).values({
      type,
      materialId,
      materialName: material.name,
      qty,
      unit: material.unit,
      user,
      workOrder: workOrder ?? null,
      note: note ?? null,
    }).returning();
    
    res.status(201).json({ ...created, createdAt: created.createdAt.toISOString() });
  } catch (err) {
    req.log.error({ err }, "Failed to create transaction");
    res.status(500).json({ error: "Failed to create transaction" });
  }
});

router.get("/category-breakdown", async (req, res) => {
  try {
    const materials = await db.select().from(materialsTable);
    const byCategory: Record<string, { count: number; lowCount: number }> = {};
    
    for (const m of materials) {
      if (!byCategory[m.category]) byCategory[m.category] = { count: 0, lowCount: 0 };
      byCategory[m.category].count++;
      const s = stockStatus(m.stock, m.minStock);
      if (s === "low" || s === "critical" || s === "out") byCategory[m.category].lowCount++;
    }
    
    const result = Object.entries(byCategory)
      .map(([category, stats]) => ({ category, ...stats }))
      .sort((a, b) => b.count - a.count);
    
    res.json(result);
  } catch (err) {
    req.log.error({ err }, "Failed to get category breakdown");
    res.status(500).json({ error: "Failed to get category breakdown" });
  }
});

                  value={search}
                  onChange={e => setSearch(e.target.value)}
                  className="pl-8"
                />
              </div>
            </div>
          </div>
        </CardHeader>
        <CardContent className="p-0">
          <Alert className="mx-6 mb-4 border-blue-200 bg-blue-50">
            <AlertTriangle className="h-4 w-4 text-blue-600" />
            <AlertTitle className="text-blue-800">How cycle count works</AlertTitle>
            <AlertDescription className="text-blue-700">
              Enter the <strong>actual physical quantity</strong> you counted on the shelf. Leave blank to skip an item. Submitted counts overwrite the system stock and create an audit transaction.
            </AlertDescription>
          </Alert>

          {isLoading ? (
            <div className="px-6 pb-6 space-y-2">
              {[...Array(8)].map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
            </div>
          ) : (
            <Table>
              <TableHeader>
                <TableRow className="bg-muted/50">
                  <TableHead className="w-28">Code</TableHead>
                  <TableHead>Material Name</TableHead>
                  <TableHead className="w-28">Category</TableHead>
                  <TableHead className="w-24 text-right">System Qty</TableHead>
                  <TableHead className="w-24 text-center">Unit</TableHead>
                  <TableHead className="w-36">Physical Count</TableHead>
                  <TableHead className="w-24 text-right">Variance</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {filteredMaterials?.map(mat => {
                  const entry = counts[mat.id];
                  const enteredVal = entry?.counted;
                  const hasValue = enteredVal !== undefined && enteredVal !== "";
                  const parsed = hasValue ? parseFloat(enteredVal!) : NaN;
                  const diff = hasValue && !isNaN(parsed) ? parsed - mat.stock : null;

                  return (
                    <TableRow key={mat.id} className={hasValue ? "bg-blue-50/50 dark:bg-blue-900/10" : ""}>
                      <TableCell className="font-mono text-xs text-muted-foreground">{mat.code}</TableCell>
                      <TableCell className="font-medium">{mat.name}</TableCell>
                      <TableCell>
                        <Badge variant="outline" className="text-xs">{mat.category}</Badge>
                      </TableCell>
                      <TableCell className="text-right font-mono">{mat.stock}</TableCell>
                      <TableCell className="text-center text-muted-foreground text-sm">{mat.unit}</TableCell>
                      <TableCell>
                        <Input
                          type="number"
                          min={0}
                          step="any"
                          placeholder="—"
                          value={enteredVal ?? ""}
                          onChange={e => setCounted(mat.id, e.target.value)}
                          className={`h-8 text-sm ${hasValue ? "border-blue-400 ring-1 ring-blue-300" : ""}`}
                        />
                      </TableCell>
                      <TableCell className="text-right">
                        {diff === null ? (
                          <span className="text-muted-foreground text-xs">—</span>
                        ) : diff === 0 ? (
                          <span className="flex items-center justify-end gap-1 text-sm text-muted-foreground">
                            <Minus className="h-3 w-3" /> 0
                          </span>
                        ) : diff > 0 ? (
                          <span className="flex items-center justify-end gap-1 text-sm text-green-600 font-semibold">
                            <ArrowUp className="h-3 w-3" /> +{diff}
                          </span>
                        ) : (
                          <span className="flex items-center justify-end gap-1 text-sm text-red-600 font-semibold">
                            <ArrowDown className="h-3 w-3" /> {diff}
                          </span>
                        )}
                      </TableCell>
                    </TableRow>
                  );
                })}
              </TableBody>
            </Table>
          )}
        </CardContent>
      </Card>

      <PrintSheetDialog
        open={showPrint}
        onClose={() => setShowPrint(false)}
        materials={materials}
      />

      <Dialog open={showResults} onOpenChange={setShowResults}>
        <DialogContent className="max-w-2xl max-h-[80vh] overflow-y-auto">
          <DialogHeader>
            <DialogTitle className="flex items-center gap-2">
              <CheckCircle2 className="h-5 w-5 text-green-600" />
              Cycle Count Complete
            </DialogTitle>
            <DialogDescription>
              {successCount} item{successCount !== 1 ? "s" : ""} processed — {adjustedCount} adjusted, {successCount - adjustedCount} confirmed correct
              {errorCount > 0 && `, ${errorCount} failed`}.
            </DialogDescription>
          </DialogHeader>

          <div className="space-y-2">
            {results?.map((r, i) => (
              <div
                key={i}
                className={`flex items-center justify-between rounded-md border px-4 py-3 text-sm ${
                  r.status === "error" ? "border-red-200 bg-red-50" :
                  r.diff === 0 ? "border-gray-200 bg-gray-50" :
                  r.diff > 0 ? "border-green-200 bg-green-50" : "border-amber-200 bg-amber-50"
                }`}
              >
                <div>
                  <span className="font-semibold">{r.materialName}</span>
                  <span className="ml-2 text-xs text-muted-foreground font-mono">{r.code}</span>
                  {r.status === "error" && (
                    <span className="ml-2 text-xs text-red-600">({r.errorMsg})</span>
                  )}
                </div>
                {r.status === "ok" ? (
                  <div className="flex items-center gap-3 text-right">
                    <span className="text-muted-foreground">{r.before} → {r.after} {r.unit}</span>
                    {r.diff === 0 ? (
                      <Badge variant="secondary" className="text-xs">No change</Badge>
                    ) : r.diff > 0 ? (
                      <Badge className="text-xs bg-green-600">+{r.diff}</Badge>
                    ) : (
                      <Badge variant="destructive" className="text-xs">{r.diff}</Badge>
                    )}
                  </div>
                ) : (
                  <Badge variant="destructive" className="text-xs">Failed</Badge>
                )}
              </div>
            ))}
          </div>

          <DialogFooter>
            <Button onClick={() => setShowResults(false)}>Done</Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  );
}
